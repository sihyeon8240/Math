\section{Euclidean Spaces}

For each $n \in \N$, the \textbf{Euclidean $n$-space} $\R^{n}$ is defined as
\[
\R^{n} \coloneqq \setbuilder*{\bm{x} = (x_{1}, x_{2}, \dots, x_{n})}{\forall k \in \N^{n}, x_{k} \in \R}
\]
where $x_{1}, x_{2}, \dots, x_{n}$ are called the \textbf{coordinates} of $\bm{x}$. The elements of $\R^{n}$
for $n > 1$ are called \textbf{points} (or \emph{vectors}) and are represented by bold symbols.

\bigskip

The operations on $\R^{n}$ are defined as
\begin{itemize}
    \item $\forall \bm{x}, \bm{y} \in \R^{n},
    \bm{x} + \bm{y} \coloneqq (x_{1} + y_{1}, x_{2} + y_{2}, \dots, x_{n} + y_{n})$
    \item $\forall \alpha \in \R, \forall \bm{x} \in \R^{n},
    \alpha \bm{x} \coloneqq (\alpha x_{1}, \alpha x_{2}, \dots, \alpha x_{n})$
\end{itemize}

\bigskip

\begin{definition} \label{an:def:inner-product-norm-distance}
    Let $\bm{x}, \bm{y} \in \R^{n}$ be given.
    \begin{itemize}
        \item The \textbf{inner product} of $\bm{x}$ and $\bm{y}$ is defined as
        \[
        \bm{x} \cdot \bm{y} \coloneqq \sum_{k = 1}^{n} x_{k}y_{k}
        \]
        \item The \textbf{norm} of $\bm{x}$ is defined as
        \[
        \norm*{\bm{x}} \coloneqq \sqrt{\bm{x} \cdot \bm{x}} = \sqrt{\sum_{k = 1}^{n} x_{k}^{2}}
        \]
        \item The \textbf{distance} between $\bm{x}$ and $\bm{y}$ is defined as
        \[
        d(\bm{x}, \bm{y}) \coloneqq \norm*{\bm{x} - \bm{y}} = \sqrt{\sum_{k = 1}^{n} (x_{k} - y_{k})^{2}}
        \]
        where one can verify that $d(x, y) = \abs*{x - y}$ in the case of $n = 1$.
    \end{itemize}
\end{definition}

\bigskip

\begin{theorem} \label{an:thm:properties-of-inner-product-norm-distance}
    Let $\bm{x}, \bm{y}, \bm{z} \in \R^{n}$ be given. Then the following properties hold:
    \begin{enumerate}[label=\textbf{(\Alph*)}]
        \item $\abs*{\bm{x} \cdot \bm{y}} \leq \norm*{\bm{x}} \cdot \norm*{\bm{y}}$
        \item $\norm*{\bm{x} + \bm{y}} \leq \norm*{\bm{x}} + \norm*{\bm{y}}$
        \item $\norm*{\bm{x} - \bm{y}} \geq \abs*{\norm*{\bm{x}} - \norm*{\bm{y}}}$
    \end{enumerate}
\end{theorem}

\begin{proof}
    \begin{enumerate}[label=\textbf{(\Alph*)}]
        \item By \Cref{an:cor:cauchy-schwarz-inequality-for-real-numbers},
        \[
        \abs*{\bm{x} \cdot \bm{y}}^{2} = \abs*{\sum_{k = 1}^{n} x_{k}y_{k}}^{2} \leq
        \sum_{k = 1}^{n} x_{k}^{2} \cdot \sum_{k = 1}^{n} y_{k}^{2}
        = \norm*{\bm{x}}^{2} \cdot \norm*{\bm{y}}^{2}
        \]
        \item By part \textbf{(A)},
        \begin{align*}
            \norm*{\bm{x} + \bm{y}}^{2} &= (\bm{x} + \bm{y}) \cdot (\bm{x} + \bm{y})
            = \norm*{\bm{x}}^{2} + 2\bm{x} \cdot \bm{y} + \norm*{\bm{y}}^{2} \\
            &\leq \norm*{\bm{x}}^{2} + 2\abs*{\bm{x} \cdot \bm{y}} + \norm*{\bm{y}}^{2}
            \leq \norm*{\bm{x}}^{2} + 2\norm*{\bm{x}} \cdot \norm*{\bm{y}} + \norm*{\bm{y}}^{2} \\
            &= (\norm*{\bm{x}} + \norm*{\bm{y}})^{2}
        \end{align*}
        \item By part \textbf{(B)},
        \[
        \begin{array}{c}
            \norm*{\bm{x}} = \norm*{(\bm{x} - \bm{y}) + \bm{y}}
            \leq \norm*{\bm{x} - \bm{y}} + \norm*{\bm{y}} \\
            \Downarrow \\
            \norm*{\bm{x}} - \norm*{\bm{y}} \leq \norm*{\bm{x} - \bm{y}}
        \end{array}
        \]
        Also,
        \[
        \begin{array}{c}
            \norm*{\bm{y}} = \norm*{(\bm{y} - \bm{x}) + \bm{x}}
            \leq \norm*{\bm{y} - \bm{x}} + \norm*{\bm{x}} \\
            \Downarrow \\
            \norm*{\bm{y}} - \norm*{\bm{x}} \leq \norm*{\bm{x} - \bm{y}}
        \end{array}
        \]
        To sum up,
        \[
        \norm*{\bm{x} - \bm{y}} \geq \abs*{\norm*{\bm{x}} - \norm*{\bm{y}}}
        \]
    \end{enumerate}
\end{proof}
