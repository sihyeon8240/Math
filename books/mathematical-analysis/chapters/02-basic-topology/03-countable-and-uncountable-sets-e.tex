\bigskip

\begin{theorem} \label{an:thm:R-is-uncountable}
    \[
    \abs*{\R} > \aleph_{0}
    \]
\end{theorem}

\begin{proof}
    By \Cref{an:lem:N-is-infinite,an:thm:subset-of-finite-set-is-finite},
    \[
    \begin{array}{c}
        \N \subseteq \R \\
        \Downarrow \\
        \abs*{\R} \geq \aleph_{0}
    \end{array}
    \]
    Define $f: \N \hookrightarrow [0, 1)$ as
    \[
    \forall n \in \N, f(n) = \frac{1}{n + 1}
    \]
    By \Cref{an:lem:N-is-infinite,an:thm:injection-from-infinite-set-implies-infinite},
    \[
    \abs*{[0, 1)} \geq \aleph_{0}
    \]
    By \Cref{an:thm:infinite-subset-of-countable-set-is-countable},
    \[
    \begin{array}{c}
        [0, 1) \subseteq \R \\
        \Downarrow \\
        \abs*{[0, 1)} > \aleph_{0} \implies \abs*{\R} > \aleph_{0}
    \end{array}
    \]
    Define $g: [0, 1) \to \prod_{n = 1}^{\infty} \set*{0, 1}$ as
    \[
    \forall x \in [0, 1), g(x) = \paren*{a_{n}^{x}}_{n = 1}^{\infty}
    \]
    where
    \[
    \paren*{a_{n}}_{n = 1}^{\infty} \text{ is the \emph{greedy base $2$ expansion} of $x$}
    \]
    Define
    \[
    Y \coloneqq g([0, 1)) \subseteq \prod_{n = 1}^{\infty} \set*{0, 1}
    \]
    By \Cref{an:thm:same-greedy-base-expansion-implies-equality,an:thm:properties-of-restriction-functions},
    \[
    \exists g|^{Y}: [0, 1) \xrightarrow{\sim} Y
    \]
    If $[0, 1)$ is countable, then
    \[
    \exists h: \N \xrightarrow{\sim} [0, 1)
    \]
    By \Cref{an:thm:properties-of-composition-functions},
    \[
    \begin{array}{c}
        g|^{Y} \circ h: \N \xrightarrow{\sim} Y \\
        \Downarrow \\
        \abs*{Y} = \aleph_{0}
    \end{array}
    \]
    Define
    \[
    Z \coloneqq \prod_{n = 1}^{\infty} \set*{0, 1} \setminus Y
    \]
    By
    \Cref{an:thm:no-real-number-matching-greedy-base-expansion,an:thm:no-real-number-matching-greedy-base-expansion-converse},
    \[
    Z = \setbuilder*{\paren*{a_{n}}_{n = 1}^{\infty} \in \prod_{n = 1}^{\infty}
    \set*{0, 1}}{\exists N \in \N, \forall n \in \N_{N}, a_{n} = 1}
    \]
    Define
    \[
    Z_{N} \coloneqq \setbuilder*{\paren*{a_{n}}_{n = 1}^{\infty} \in
    \prod_{n = 1}^{\infty} \set*{0, 1}}{\forall n \in \N_{N}, a_{n} = 1}
    \]
    for each $N \in \N$. Define $F_{N}: \prod_{n = 1}^{N - 1} \set*{0, 1} \twoheadrightarrow Z_{N}$ as
    \[
    \forall \paren*{a_{n}}_{n = 1}^{N - 1} \in \prod_{n = 1}^{N - 1} \set*{0, 1},
    F_{N}\paren*{\paren*{a_{n}}_{n = 1}^{N - 1}} = \paren*{a_{n}}_{n = 1}^{N - 1} + \paren*{1}_{n = N}^{\infty}
    \]
    By \Cref{an:thm:product-of-finite-sets-is-finite,an:thm:surjection-from-finite-set-implies-finite},
    \[
    \forall N \in \N, 0 < \abs*{Z_{N}} < \aleph_{0}
    \]
    By \Cref{an:thm:countable-union-of-finite-sets-is-countable},
    \[
    \begin{array}{c}
        Z = \bigcup_{N = 1}^{\infty} Z_{N} \\
        \Downarrow \\
        \abs*{Z} = \aleph_{0}
    \end{array}
    \]
    By \Cref{an:cor:finite-union-of-countable-sets-is-countable,an:lem:infinite-product-of-finite-sets-is-uncountable},
    \[
    \begin{array}{c}
        \prod_{n = 1}^{\infty} \set*{0, 1} = Y \cup Z \\
        \Downarrow \\
        \abs*{\prod_{n = 1}^{\infty} \set*{0, 1}} = \aleph_{0} \\
        \Downarrow \\
        \bot
    \end{array}
    \]
    Therefore,
    \[
    \begin{array}{c}
        \abs*{[0, 1)} > \aleph_{0} \\
        \Downarrow \\
        \abs*{\R} > \aleph_{0}
    \end{array}
    \]
\end{proof}
