\begin{proof}
    By \Cref{an:cor:closed-set-complement-is-open-set}, $(E_{k})^{c}$ is open for each $k = 1, 2, \dots, n$.
    Then, by \Cref{an:thm:finite-intersection-of-open-sets-is-open}, we have
    \[
    \bigcap_{k = 1}^{n} (E_{k})^{c} \text{ is open.}
    \]
    By \Cref{an:thm:open-set-complement-is-closed-set}, we have
    \[
    \bigcup_{k = 1}^{n} E_{k} = \paren*{\bigcap_{k = 1}^{n} (E_{k})^{c}}^{c} \text{ is closed.}
    \]
\end{proof}

\begin{definition} \label{an:def:closure}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Let $E'$ be the set of all limit points of $E$.
    The \textbf{closure} of $E$, which is denoted by $\overline{E}$, is defined as
    \[
    \overline{E} \coloneqq E \cup E'.
    \]
\end{definition}

\bigskip

\begin{theorem} \label{an:thm:properties-of-closure}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then we have the following properties:
    \begin{enumerate}[label=\textbf{(\Alph*)}]
        \item $\overline{E}$ is closed.
        \item $E = \overline{E} \iff E$ is closed.
        \item For every closed subset $F \subseteq X$ such that $E \subseteq F$, we have $\overline{E} \subseteq F$.
    \end{enumerate}
    \textbf{(C)} states that the closure of $E$ is the smallest closed set containing $E$.
\end{theorem}

\begin{proof}
    \begin{enumerate}[label=\textbf{(\Alph*)}]
        \item We will show that $\paren*{\overline{E}}^{c}$ is open. Let $p \in \paren*{\overline{E}}^{c}$ be given.
        Since $p \notin E$ and $p \notin E'$, there exists some $r \in \R^{+}$ such that
        \[
        E \cap \widetilde{N}_{r}(p) = \vnot \implies E \cap N_{r}(p) = \vnot \implies N_{r}(p) \subseteq E^{c}.
        \]
        By \Cref{an:thm:neighborhood-is-open}, $N_{r}(p)$ is open. Then, for every $q \in N_{r}(p)$,
        there exists $\veps \in \R^{+}$ such that
        \begin{align*}
            N_{\veps}(q) \subseteq N_{r}(p) \subseteq E^{c}
            \implies &E \cap N_{\veps}(q) = \vnot \\
            \implies &E \cap \paren*{\widetilde{N}_{\veps}(q) \cup \set*{q}} = \vnot \\
            \implies &\paren*{E \cap \widetilde{N}_{\veps}(q)} \cup \paren*{E \cap \set*{q}} = \vnot \\
            \implies &E \cap \widetilde{N}_{\veps}(q) = \vnot \\
            \implies &q \notin E'.
        \end{align*}
        Since $q$ is arbitrary, we have
        \[
        N_{r}(p) \subseteq \paren*{E'}^{c} \implies N_{r}(p) \subseteq
        \paren*{\overline{E}}^{c} = E^{c} \cap \paren*{E'}^{c},
        \]
        which implies that $p$ is an interior point of $\paren*{\overline{E}}^{c}$. Since $p$ is arbitrary, $\paren*{
        \overline{E}}^{c}$ is open, which implies that $\overline{E}$ is closed by \Cref{an:cor:closed-set-complement-is-open-set}.
        \item $E$ is closed $\iff E' \subseteq E \iff E \cup E' = E \iff \overline{E} = E$.
        \item Let $F \subseteq X$ be a closed set such that $E \subseteq F$. Let $p \in E'$ be given.
        Then, for every $r \in \R^{+}$, we have
        \[
        E \cap \widetilde{N}_{r}(p) \neq \vnot \implies F \cap \widetilde{N}_{r}(p) \neq \vnot,
        \]
        which implies that $p \in F' \subseteq F$. Therefore, we have $E' \subseteq F$.
        Since $E \subseteq F$, we have $\overline{E} = E \cup E' \subseteq F$.
    \end{enumerate}
\end{proof}

\begin{theorem} \label{an:thm:supremum-in-closure}
    Let $\paren*{\R, \norm*{\cdot}}$ be given. Let $E \subseteq \R$ be nonempty and bounded above. Then we have
    \[
    \sup E \in \overline{E},
    \]
    which implies that
    \[
    E \text{ is closed} \implies \sup E \in E.
    \]
\end{theorem}

\begin{proof}
    By the \emph{least upper bound property},
    there exists $\alpha = \sup E \in \R$. Assume that $\alpha \in E$. Then we have
    \[
    E \subseteq \overline{E} \implies \alpha \in \overline{E}.
    \]
    Otherwise, for every $x \in \R^{+}$, $\alpha - x$ is not an upper bound of $E$.
    Thus there exists some $p \in E$ such that
    \[
    \alpha - x < p < \alpha \implies p \in \widetilde{N}_{x}(\alpha) \cap E,
    \]
    which implies that $\alpha \in E'$. Therefore, we have
    \[
    E' \subseteq \overline{E} \implies \alpha \in \overline{E}.
    \]
\end{proof}

\begin{theorem} \label{an:thm:isolated-points}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then we have
    \[
    E \setminus E' = \setbuilder*{p \in X}{p \text{ is an isolated point of } E}.
    \]
\end{theorem}

\begin{proof}
    \textbf{($\subseteq$)} Let $p \in E \setminus E'$ be given.
    Since $p \notin E'$, there exists some $r \in \R^{+}$ such that
    \[
    E \cap \widetilde{N}_{r}(p) = \vnot \implies \paren*{E \cap N_{r}(p)} \setminus \set*{p} = \vnot.
    \]
    Since $p \in E$ and $p \in N_{r}(p)$, we have
    \[
    E \cap N_{r}(p) = \set*{p},
    \]
    which implies that $p$ is an isolated point of $E$.
    \\[6mm]
    \textbf{($\supseteq$)} Let $p$ be an isolated point of $E$. Then there exists some $r \in \R^{+}$ such that
    \[
    E \cap N_{r}(p) = \set*{p}.
    \]
    Then we have
    \begin{itemize}
        \item $E \cap \widetilde{N}_{r}(p) = \vnot \implies p \notin E'$.
        \item $p \in E$.
    \end{itemize}
    Therefore, we have $p \in E \setminus E'$.
\end{proof}

    Let $(X, d)$ be a metric space and $E \subseteq X$. The \textbf{boundary} of $E$,
    which is denoted by $\partial E$, is defined as
    \[
    \partial E \coloneqq \overline{E} \cap \overline{E^{c}}.

\end{definition}

\begin{lemma} \label{an:lem:deleted-neighborhood-in-rn-is-non-empty}
    Let $(\R^{n}, \norm*{\cdot})$ be given for $n \in \N$. Then we have
    \[
    \widetilde{N}_r(\bm{p}) \neq \vnot \quad \text{for all } \bm{p} \in \R^{n} \text{ and } r \in \R^{+}.
    \]
\end{lemma}

\begin{proof}
    Let $\bm{p} = (p_{1}, p_{2}, \dots, p_{n}) \in \R^{n}$ and $r \in \R^{+}$ be given. Then we have
    \[
    d\paren*{\bm{p} + \frac{r}{2} \bm{e_{1}}, \bm{p}} = \norm*{\paren*{\bm{p} + \frac{r}{2} \bm{e_{1}}}
    - \bm{p}} = \norm*{\frac{r}{2} \bm{e_{1}}} = \frac{r}{2} \norm*{\bm{e_{1}}} = \frac{r}{2} < r.
    \]
    where $\bm{e_{1}} = (1, 0, 0, \dots, 0) \in \R^{n}$. Therefore, we have
    \[
    \bm{p} \neq \paren*{\bm{p} + \frac{r}{2} \bm{e_{1}}} \in \widetilde{N}_r(\bm{p}) \implies \widetilde{N}_r(\bm{p}) \neq \vnot.
    \]
\end{proof}
