\begin{proof}
    By \Cref{an:cor:closed-set-complement-is-open-set},
    \[
    \forall k \in \N^{n}, (E_{k})^{c} \text{ is open}
    \]
    By \Cref{an:thm:finite-intersection-of-open-sets-is-open},
    \[
    \bigcap_{k = 1}^{n} (E_{k})^{c} \text{ is open}
    \]
    By \Cref{an:thm:open-set-complement-is-closed-set},
    \[
    \bigcup_{k = 1}^{n} E_{k} = \paren*{\bigcap_{k = 1}^{n} (E_{k})^{c}}^{c} \text{ is closed}
    \]
\end{proof}

\begin{definition} \label{an:def:closure}
    Let $(X, d)$ be a metric space and $E \subseteq X$. The \textbf{closure} $\overline{E}$ of $E$ is defined as
    \[
    \overline{E} \coloneqq E \cup E'
    \]
\end{definition}

\bigskip

\begin{theorem} \label{an:thm:properties-of-closure}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then
    \begin{enumerate}[label=\textbf{(\Alph*)}]
        \item $\overline{E}$ is closed
        \item $E = \overline{E} \iff E$ is closed
        \item If $F \subseteq X$ satisfies
        \begin{itemize}
            \item $E \subseteq F$
            \item $F$ is closed
        \end{itemize}
        then $\overline{E} \subseteq F$
    \end{enumerate}
\end{theorem}

\begin{proof}
    \begin{enumerate}[label=\textbf{(\Alph*)}]
        \item Let $p \in \paren*{\overline{E}}^{c}$ be given. Then
        \[
        \begin{array}{c}
            p \notin E \land p \notin E' \\
            \Downarrow \\
            \exists r \in \R^{+}, E \cap \widetilde{N}_{r}(p) = \vnot \\
            \Downarrow \\
            E \cap N_{r}(p) = \vnot \\
            \Downarrow \\
            N_{r}(p) \subseteq E^{c}
        \end{array}
        \]
        By \Cref{an:thm:neighborhood-is-open}, $N_{r}(p)$ is open. Then
        \[
        \begin{array}{c}
            \forall q \in N_{r}(p), \exists \veps \in \R^{+}, N_{\veps}(q) \subseteq N_{r}(p) \subseteq E^{c} \\
            \Downarrow \\
            E \cap N_{\veps}(q) = \vnot \\
            \Downarrow \\
            E \cap \widetilde{N}_{\veps}(q) = \vnot \\
            \Downarrow \\
            q \notin E'
        \end{array}
        \]
        Since $q$ is arbitrary,
        \[
        \begin{array}{c}
            N_{r}(p) \subseteq \paren*{E'}^{c} \\
            \Downarrow \\
            N_{r}(p) \subseteq \paren*{\overline{E}}^{c} = E^{c} \cap \paren*{E'}^{c} \\
            \Downarrow \\
            p \text{ is an interior point of } \paren*{\overline{E}}^{c}
        \end{array}
        \]
        Since $p$ is arbitrary,
        \[
        \paren*{\overline{E}}^{c} \text{ is open}
        \]
        By \Cref{an:cor:closed-set-complement-is-open-set},
        \[
        \overline{E} \text{ is closed}
        \]
        \item $E$ is closed $\iff E' \subseteq E \iff E \cup E' = E \iff \overline{E} = E$
        \item Let $F \subseteq X$ satisfy
        \begin{itemize}
            \item $E \subseteq F$
            \item $F$ is closed
        \end{itemize}
        Let $p \in E'$ be given. Then,
        \[
        \begin{array}{c}
            \forall r \in \R^{+}, E \cap \widetilde{N}_{r}(p) \neq \vnot \\
            \Downarrow \\
            F \cap \widetilde{N}_{r}(p) \neq \vnot \\
            \Downarrow \\
            p \in F' \subseteq F
        \end{array}
        \]
        Since $p$ is arbitrary,
        \[
        \begin{array}{c}
            E' \subseteq F' \subseteq F \\
            \Downarrow \\
            \overline{E} = E \cup E' \subseteq F
        \end{array}
        \]
    \end{enumerate}
\end{proof}

\begin{theorem} \label{an:thm:supremum-in-closure}
    Let $\paren*{\R, \norm*{\cdot}}$ be given. If $E \subseteq \R$ satisfies
    \begin{itemize}
        \item $E \neq \vnot$
        \item $E$ is bounded above
    \end{itemize}
    then
    \[
    \exists \sup E \in \overline{E}
    \]
    Also,
    \[
    E \text{ is closed} \implies \exists \sup E \in E
    \]
\end{theorem}

\begin{proof}
    By the \emph{least upper bound property},
    \[
    \exists \alpha \in \R, \alpha = \sup E
    \]
    If $\alpha \in E$, then
    \[
    \begin{array}{c}
        E \subseteq \overline{E} \\
        \Downarrow \\
        \alpha \in \overline{E}
    \end{array}
    \]
    Otherwise,
    \[
    \begin{array}{c}
        \forall x \in \R^{+}, \alpha - x \text{ is not an upper bound of } E \\
        \Downarrow \\
        \exists p \in E, \alpha - x < p < \alpha \\
        \Downarrow \\
        p \in \widetilde{N}_{x}(\alpha) \cap E \\
        \Downarrow \\
        \alpha \in E'
    \end{array}
    \]
    Therefore,
    \[
    \begin{array}{c}
        E' \subseteq \overline{E} \\
        \Downarrow \\
        \alpha \in \overline{E}
    \end{array}
    \]
\end{proof}

\begin{theorem} \label{an:thm:isolated-points}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then
    \[
    E \setminus E' = \setbuilder*{p \in X}{p \text{ is an isolated point of } E}
    \]
\end{theorem}

\begin{proof}
    \textbf{($\subseteq$)} Let $p \in E \setminus E'$ be given. Then
    \[
    \begin{array}{c}
        p \notin E' \\
        \Downarrow \\
        \exists r \in \R^{+}, E \cap \widetilde{N}_{r}(p) = \vnot \\
        \Downarrow \\
        \paren*{E \cap N_{r}(p)} \setminus \set*{p} = \vnot
    \end{array}
    \]
    Therefore,
    \[
    \begin{array}{c}
        p \in E \land p \in N_{r}(p) \\
        \Downarrow \\
        E \cap N_{r}(p) = \set*{p} \\
        \Downarrow \\
        p \text{ is an isolated point of } E
    \end{array}
    \]
    \textbf{($\supseteq$)} Let $p$ be an isolated point of $E$. Then
    \[
    \exists r \in \R^{+}, E \cap N_{r}(p) = \set*{p}
    \]
    Therefore,
    \[
    \begin{array}{c}
        p \in E \\
        \Downarrow \\
        E \cap \widetilde{N}_{r}(p) = \vnot \\
        \Downarrow \\
        p \notin E' \\
        \Downarrow \\
        p \in E \setminus E'
    \end{array}
    \]
\end{proof}
