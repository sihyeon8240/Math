\begin{theorem} \label{an:thm:open-set-complement-is-closed-set}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then
    \[
    E \text{ is open} \iff E^{c} \text{ is closed}
    \]
\end{theorem}

\begin{proof}
    \textbf{($\implies$)} Let $E$ be open. Let $p \in \paren*{E^{c}}'$ be given. Then
    \[
    \forall r \in \R^{+}, E^{c} \cap \widetilde{N}_{r}(p) \neq \vnot
    \]
    If $p \in E$, then
    \[
    \begin{array}{c}
        p \in E \\
        \Downarrow \\
        \exists \veps \in \R^{+}, N_{\veps}(p) \subseteq E \\
        \Downarrow \\
        E^{c} \cap \widetilde{N}_{\veps}(p) = \vnot \\
        \Downarrow \\
        \bot
    \end{array}
    \]
    Therefore,
    \[
    \begin{array}{c}
        p \notin E \\
        \Downarrow \\
        p \in E^{c}
    \end{array}
    \]
    Since $p$ is arbitrary,
    \[
    \begin{array}{c}
        \paren*{E^{c}}' \subseteq E^{c} \\
        \Downarrow \\
        E^{c} \text{ is closed}
    \end{array}
    \]
    \textbf{($\impliedby$)} Let $E^{c}$ be closed. Let $p \in E$ be given. If
    \[
    \nexists r \in \R^{+}, N_{r}(p) \subseteq E,
    \]
    then
    \[
    \forall r \in \R^{+}, E^{c} \cap N_{r}(p) \neq \vnot
    \]
    Since $p \in E$,
    \[
    \begin{array}{c}
        p \notin E^{c} \\
        \Downarrow \\
        E^{c} \cap \widetilde{N}_{r}(p) \neq \vnot \\
        \Downarrow \\
        p \in \paren*{E^{c}}'
    \end{array}
    \]
    Since $E^{c}$ is closed,
    \[
    \begin{array}{c}
        p \in \paren*{E^{c}}' \\
        \Downarrow \\
        p \in E^{c} \\
        \Downarrow \\
        \bot
    \end{array}
    \]
    Therefore,
    \[
    \begin{array}{c}
        \exists r \in \R^{+}, N_{r}(p) \subseteq E \\
        \Downarrow \\
        p \text{ is an interior point of } E
    \end{array}
    \]
    Since $p$ is arbitrary, $E$ is open.
\end{proof}

\bigskip

\begin{corollary} \label{an:cor:closed-set-complement-is-open-set}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then
    \[
    E \text{ is closed} \iff E^{c} \text{ is open}
    \]
\end{corollary}

\begin{theorem} \label{an:thm:arbitrary-union-of-open-sets-is-open}
    Let $(X, d)$ be a metric space.
    \[
    \forall Y, \paren*{\forall y \in Y, E_{y} \subseteq X \text{ is open}} \implies \bigcup_{y \in Y} E_{y} \text{ is open}
    \]
\end{theorem}

\begin{proof}
    Let $p \in \bigcup_{y \in Y} E_{y}$ be given. Then
    \[
    \exists y_{0} \in Y, p \in E_{y_{0}}
    \]
    Since $E_{y_{0}}$ is open,
    \[
    \begin{array}{c}
        \exists r \in \R^{+}, N_{r}(p) \subseteq E_{y_{0}} \\
        \Downarrow \\
        N_{r}(p) \subseteq E_{y_{0}} \subseteq \bigcup_{y \in Y} E_{y} \\
        \Downarrow \\
        p \text{ is an interior point of } \bigcup_{y \in Y} E_{y}
    \end{array}
    \]
    Since $p$ is arbitrary, $\bigcup_{y \in Y} E_{y}$ is open.
\end{proof}

\bigskip

\begin{corollary} \label{an:cor:arbitrary-intersection-of-closed-sets-is-closed}
    Let $(X, d)$ be a metric space.
    \[
    \forall Y, \paren*{\forall y \in Y, E_{y} \subseteq X \text{ is closed}} \implies \bigcap_{y \in Y} E_{y} \text{ is closed}
    \]
\end{corollary}

\begin{proof}
    By \Cref{an:cor:closed-set-complement-is-open-set},
    \[
    \forall y \in Y, (E_{y})^{c} \text{ is open}
    \]
    By \Cref{an:thm:arbitrary-union-of-open-sets-is-open},
    \[
    \bigcup_{y \in Y} (E_{y})^{c} \text{ is open}
    \]
    By \Cref{an:thm:open-set-complement-is-closed-set},
    \[
    \bigcap_{y \in Y} E_{y} = \paren*{\bigcup_{y \in Y} (E_{y})^{c}}^{c} \text{ is closed}
    \]
\end{proof}

\begin{theorem} \label{an:thm:finite-intersection-of-open-sets-is-open}
    Let $(X, d)$ be a metric space.
    \[
    \paren*{\forall k \in \N^{n}, E_{k} \subseteq X \text{ is open}} \implies \bigcap_{k = 1}^{n} E_{k} \text{ is open}
    \]
\end{theorem}

\begin{proof}
    Let $p \in \bigcap_{k = 1}^{n} E_{k}$ be given. Then
    \[
    \forall j \in \N^{n}, p \in E_{j}
    \]
    Since each $E_{j}$ is open,
    \[
    \exists r_{j} \in \R^{+}, N_{r_{j}}(p) \subseteq E_{j}
    \]
    for each $j \in \N^{n}$. Define
    \[
    R \coloneqq \setbuilder*{r_{j}}{j \in \N^{n}} \subseteq \R^{+}
    \]
    Define $f: \N^{n} \twoheadrightarrow R$ as
    \[
    f(j) = r_{j}
    \]
    By \Cref{an:lem:subset-of-Nn-is-finite,an:thm:surjection-from-finite-set-implies-finite},
    \[
    \abs*{R} < \aleph_{0}
    \]
    By \Cref{an:thm:finite-ordered-set-has-extremum},
    \[
    \exists r \in \R^{+}, r = \min R
    \]
    Then
    \[
    \begin{array}{c}
        \forall j \in \N^{n}, N_{r}(p) \subseteq N_{r_{j}}(p) \subseteq E_{j} \\
        \Downarrow \\
        N_{r}(p) \subseteq \bigcap_{k = 1}^{n} E_{k} \\
        \Downarrow \\
        p \text{ is an interior point of } \bigcap_{k = 1}^{n} E_{k}
    \end{array}
    \]
    Since $p$ is arbitrary, $\bigcap_{k = 1}^{n} E_{k}$ is open.
\end{proof}

\bigskip

\begin{corollary} \label{an:cor:finite-union-of-closed-sets-is-closed}
    Let $(X, d)$ be a metric space.
    \[
    \paren*{\forall k \in \N^{n}, E_{k} \subseteq X \text{ is closed}} \implies \bigcup_{k = 1}^{n} E_{k} \text{ is closed}
    \]
\end{corollary}
