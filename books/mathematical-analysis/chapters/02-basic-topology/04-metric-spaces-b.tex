\begin{theorem} \label{an:thm:open-set-complement-is-closed-set}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then
    \[
    E \text{ is open} \iff E^{c} \text{ is closed}
    \]
\end{theorem}

\begin{proof}
    \textbf{($\implies$)} Let $E$ be open. Let $p \in \paren*{E^{c}}'$ be given. Then
    \[
    \forall r \in \R^{+}, E^{c} \cap \widetilde{N}_{r}(p) \neq \vnot
    \]
    If $p \in E$, then
    \[
    \begin{array}{c}
        p \in E \\
        \Downarrow \\
        \exists \veps \in \R^{+}, N_{\veps}(p) \subseteq E \\
        \Downarrow \\
        E^{c} \cap \widetilde{N}_{\veps}(p) = \vnot \\
        \Downarrow \\
        \bot
    \end{array}
    \]
    Therefore,
    \[
    \begin{array}{c}
        p \notin E \\
        \Downarrow \\
        p \in E^{c}
    \end{array}
    \]
    Since $p$ is arbitrary,
    \[
    \begin{array}{c}
        \paren*{E^{c}}' \subseteq E^{c} \\
        \Downarrow \\
        E^{c} \text{ is closed}
    \end{array}
    \]
    \textbf{($\impliedby$)} Let $E^{c}$ be closed. Let $p \in E$ be given. If
    \[
    \nexists r \in \R^{+}, N_{r}(p) \subseteq E,
    \]
    then
    \[
    \forall r \in \R^{+}, E^{c} \cap N_{r}(p) \neq \vnot
    \]
    Since $p \in E$,
    \[
    \begin{array}{c}
        p \notin E^{c} \\
        \Downarrow \\
        E^{c} \cap \widetilde{N}_{r}(p) \neq \vnot \\
        \Downarrow \\
        p \in \paren*{E^{c}}'
    \end{array}
    \]
    Since $E^{c}$ is closed,
    \[
    \begin{array}{c}
        p \in \paren*{E^{c}}' \\
        \Downarrow \\
        p \in E^{c} \\
        \Downarrow \\
        \bot
    \end{array}
    \]
    Therefore,
    \[
    \begin{array}{c}
        \exists r \in \R^{+}, N_{r}(p) \subseteq E \\
        \Downarrow \\
        p \text{ is an interior point of } E
    \end{array}
    \]
    Since $p$ is arbitrary, $E$ is open.
\end{proof}

\bigskip

\begin{corollary} \label{an:cor:closed-set-complement-is-open-set}
    Let $(X, d)$ be a metric space and $E \subseteq X$. Then
    \[
    E \text{ is closed} \iff E^{c} \text{ is open}
    \]
\end{corollary}

\begin{theorem} \label{an:thm:arbitrary-union-of-open-sets-is-open}
    Let $(X, d)$ be a metric space. Let $E_{y} \subseteq X$ be an open set for each $y \in Y$. Then we have
    \[
    \bigcup_{y \in Y} E_{y} \text{ is open}
    \]
    whether $Y$ is finite, countable, or uncountable. This theorem states that an arbitrary union of open sets is open.
\end{theorem}

\begin{proof}
    Let $p \in \bigcup_{y \in Y} E_{y}$ be given. Then there exists some $y_{0} \in Y$
    such that $p \in E_{y_{0}}$. Since $E_{y_{0}}$ is open, there exists some $r \in \R^{+}$
    such that $N_{r}(p) \subseteq E_{y_{0}}$. Then we have
    \[
    N_{r}(p) \subseteq E_{y_{0}} \subseteq \bigcup_{y \in Y} E_{y},
    \]
    which implies that $p$ is an interior point of $\bigcup_{y \in Y} E_{y}$.
    Since $p$ is arbitrary, $\bigcup_{y \in Y} E_{y}$ is open.
\end{proof}

\begin{corollary} \label{an:cor:arbitrary-intersection-of-closed-sets-is-closed}
    Let $(X, d)$ be a metric space. Let $E_{y} \subseteq X$ be a closed set for each $y \in Y$. Then we have
    \[
    \bigcap_{y \in Y} E_{y} \text{ is closed}
    \]
    whether $Y$ is finite, countable, or uncountable. This corollary states
    that an arbitrary intersection of closed sets is closed.
\end{corollary}

\begin{proof}
    By \Cref{an:cor:closed-set-complement-is-open-set}, $(E_{y})^{c}$ is open for each $y \in Y$.
    Then, by \Cref{an:thm:arbitrary-union-of-open-sets-is-open}, we have
    \[
    \bigcup_{y \in Y} (E_{y})^{c} \text{ is open.}
    \]
    By \Cref{an:thm:open-set-complement-is-closed-set}, we have
    \[
    \bigcap_{y \in Y} E_{y} = \paren*{\bigcup_{y \in Y} (E_{y})^{c}}^{c} \text{ is closed.}
    \]
\end{proof}

\begin{theorem} \label{an:thm:finite-intersection-of-open-sets-is-open}
    Let $(X, d)$ be a metric space. Let $E_{1}, E_{2}, \dots, E_{n} \subseteq X$
    be open sets for $n \in \N$. Then we have
    \[
    \bigcap_{k = 1}^{n} E_{k} \text{ is open.}
    \]
    This theorem states that a finite intersection of open sets is open.
\end{theorem}

\begin{proof}
    Let $p \in \bigcap_{k = 1}^{n} E_{k}$ be given. Then we have $p \in E_{j}$ for all $j = 1, 2, \dots, n$.
    Since $E_{j}$ is open, there exists some $r_{j} \in \R^{+}$ such that $N_{r_{j}}(p) \subseteq E_{j}$ for each $j =
    1, 2, \dots, n$. Let $R$ be a set defined as
    \[
    R \coloneqq \setbuilder*{r_{j}}{j = 1, 2, \dots, n} \subseteq \R^{+}.
    \]
    Let $f: \N^{n} \to R$ be a function defined as
    \[
    f(j) = r_{j}.
    \]
    Then $f$ is surjective. By \Cref{an:lem:subset-of-Nn-is-finite,an:thm:surjection-from-finite-set-implies-finite}, $R$ is finite.
    Then, by \Cref{an:thm:finite-ordered-set-has-extremum}, $R$ has a minimum element, say $r = \min R > 0$. Then we have
    \[
    N_{r}(p) \subseteq N_{r_{j}}(p) \subseteq E_{j} \text{ for all } j = 1, 2, \dots, n
    \implies N_{r}(p) \subseteq \bigcap_{k = 1}^{n} E_{k},
    \]
    which implies that $p$ is an interior point of $\bigcap_{k = 1}^{n} E_{k}$.
    Since $p$ is arbitrary, $\bigcap_{k = 1}^{n} E_{k}$ is open.
\end{proof}
